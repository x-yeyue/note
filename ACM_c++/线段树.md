# 基本概述

**可以维护的信息:** 累加和、最大值、最小值。(父节点的信息可以用 $O(1)$ 的时间，从子节点得到)

经典功能(单次调用时间复杂度 $O(\log n)$):
- **范围查询**，获得范围内累加和、最大值、最小值等。
- **范围修改**，修改范围内每个数。

# 区间修改(加)、累加和

```c++
// 线段树-区间和
struct Segment_tree{
    int n; // 给定数组元素个数
    // tree[] := 线段树，区间和
    // lazy[] := 懒惰标记
    vector<ll> tree, lazy;

    // 初始化线段树的大小
    Segment_tree(int _n){
        n = _n;
        tree.assign(n << 2, 0);
        lazy.assign(n << 2, 0);
    }

    // 构建线段树
    void build(int idx, int l, int r, const vector<ll>& arr){
        if(l == r){
            tree[idx] = arr[l];
            return;
        }
        int mid = l + ((r - l) >> 1);
        build(idx << 1, l, mid, arr);
        build(idx << 1 | 1, mid + 1, r, arr);
        tree[idx] = tree[idx << 1] + tree[idx << 1 | 1];    
    }

	// 添加懒惰标记
    void add_tag(int idx, int l, int r, ll val){
        lazy[idx] += val;
        tree[idx] += (r - l + 1) * val;
    }

    // 传递懒惰标记
    void push_down(int idx, int l, int r){
        if(!lazy[idx]) return;
        int mid = l + ((r - l) >> 1);
        add_tag(idx << 1, l, mid, lazy[idx]);
        add_tag(idx << 1 | 1, mid + 1, r, lazy[idx]);
        lazy[idx] = 0;
    }

    // 查询区间和
    ll query(int idx, int l, int r, int lf, int rt){
        if(lf <= l && r <= rt){
            return tree[idx];
        }
        push_down(idx, l, r);
        int mid = l + ((r - l) >> 1);
        ll ans = 0;
        if(lf <= mid){
            ans += query(idx << 1, l, mid, lf, rt);
        }
        if(mid < rt){
            ans += query(idx << 1 | 1, mid + 1, r, lf, rt);
        }
        return ans;
    }

    // 区间修改-加
    void update(int idx, int l, int r, int lf, int rt, ll val){
        if(lf <= l && r <= rt){
            add_tag(idx, l, r, val);
            return;
        }
        push_down(idx, l, r);
        int mid = l + ((r - l) >> 1);
        if(lf <= mid){
            update(idx << 1, l, mid, lf, rt, val);
        }
        if(mid < rt){
            update(idx << 1 | 1, mid + 1, r, lf, rt, val);
        }
        tree[idx] = tree[idx << 1] + tree[idx << 1 | 1];
    }
};
```

# 区间修改(加、重置)、累加和

```c++
struct Segment_tree{
    int n; // 给定数组元素个数
    vector<ll> tree; // 区间和
    vector<ll> lazy_add; // 添加的懒惰标记
    vector<ll> lazy_set; // 重置的懒惰标记
    vector<bool> is_set; // 是否重置
    
    // 初始化线段树的大小
    Segment_tree(int _n){
        n = _n;
        tree.assign(n << 2, 0);
        lazy_add.assign(n << 2, 0);
        lazy_set.assign(n << 2, 0);
        is_set.assign(n << 2, false);
    }
	
	// 构建线段树
    void build(int idx, int l, int r, const vector<ll> &arr){
        if(l == r){
            tree[idx] = arr[l];
            return;
        }
        int mid = l + ((r - l) >> 1);
        build(idx << 1, l, mid, arr);
        build(idx << 1 | 1, mid + 1, r, arr);
        tree[idx] = tree[idx << 1] + tree[idx << 1 | 1];
    }
	
	// 添加 加法 懒惰标记
    void add_tag(int idx, int l, int r, ll val){
        tree[idx] += (r - l + 1) * val;
        lazy_add[idx] += val;
    }
	
	// 添加 重置 懒惰标记
    void set_tag(int idx, int l, int r, ll val){
        tree[idx] = (r - l + 1) * val;
        lazy_set[idx] = val;
        is_set[idx] = true;
        lazy_add[idx] = 0;
    }
    
	// 传递懒惰标记
    void push_down(int idx, int l, int r){
        int mid = l + ((r - l) >> 1);
        if(is_set[idx]){
            set_tag(idx << 1, l, mid, lazy_set[idx]);
            set_tag(idx << 1 | 1, mid + 1, r, lazy_set[idx]);
            is_set[idx] = false;
        }
        if(lazy_add[idx]){
            add_tag(idx << 1, l, mid, lazy_add[idx]);
            add_tag(idx << 1 | 1, mid + 1, r, lazy_add[idx]);
            lazy_add[idx] = 0;
        }
    }
	
	// 查询区间累加和
    ll query(int idx, int l, int r, int lf, int rt){
        if(lf <= l && r <= rt) return tree[idx];
        push_down(idx, l, r);
        int mid = l + ((r - l) >> 1);
        ll ans = 0;
        if(lf <= mid) ans += query(idx << 1, l, mid, lf, rt);
        if(mid < rt) ans += query(idx << 1 | 1, mid + 1, r, lf, rt);
        return ans;
    }
	
	// 区间修改-加
    void update_add(int idx, int l, int r, int lf, int rt, ll val){
        if(lf <= l && r <= rt){
            add_tag(idx, l, r, val);
            return;
        }
        push_down(idx, l, r);
        int mid = l + ((r - l) >> 1);
        if(lf <= mid) update_add(idx << 1, l, mid, lf, rt, val);
        if(mid < rt) update_add(idx << 1 | 1, mid + 1, r, lf, rt, val);
        tree[idx] = tree[idx << 1] + tree[idx << 1 | 1];
    }
	
	// 区间修改-重置
    void update_set(int idx, int l, int r, int lf, int rt, ll val){
        if(lf <= l && r <= rt){
            set_tag(idx, l, r, val);
            return;
        }
        push_down(idx, l, r);
        int mid = l + ((r - l) >> 1);
        if(lf <= mid) update_set(idx << 1, l, mid, lf, rt, val);
        if(mid < rt) update_set(idx << 1 | 1, mid + 1, r, lf, rt, val);
        tree[idx] = tree[idx << 1] + tree[idx << 1 | 1];
    }
};
```
